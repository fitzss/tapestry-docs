Chat Interface for Robotic Tasks

Overview

This guide demonstrates how to create a chat-like interface for managing robotic tasks, integrating with our walking skeleton project.

    B --> C[Constraint Checker]
    C --> D[Task Validator]

    E --> F[Physical System]
    D --> G[User Feedback]

System Components

1. Task Parser

class TaskParser:
    def parse_command(self, text_input):
        """
        Parse natural language commands into structured tasks
        """
        task = {
            'action': None,
            'parameters': {}
        }

        # Parse lifting tasks
        if 'lift' in text_input:
            weight_match = re.search(r'(\d+)kg', text_input)
            height_match = re.search(r'(\d+)m', text_input)

            task['action'] = 'lift'
            task['parameters'] = {
                'weight': int(weight_match.group(1)) if weight_match else None,
                'height': float(height_match.group(1)) if height_match else 1.0
            }

        # Parse walking tasks
        elif 'walk' in text_input:
            task['action'] = 'walk'
            task['parameters'] = {
                'destination': extract_destination(text_input),
                'speed': extract_speed(text_input)
            }

        return task

2. Constraint Validator

class ConstraintValidator:
    def __init__(self, robot_capabilities):
        self.capabilities = robot_capabilities

    def validate_task(self, task):
        if task['action'] == 'lift':
            return self.check_lift_constraints(task['parameters'])
        elif task['action'] == 'walk':
            return self.check_walk_constraints(task['parameters'])

    def check_lift_constraints(self, params):
        if params['weight'] > self.capabilities['max_lift']:
            return {
                'possible': False,
                'reason': f"Weight exceeds maximum capacity of {self.capabilities['max_lift']}kg"
            }
        return {'possible': True}

3. Task Queue Manager

class TaskQueue:
    def __init__(self):
        self.queue = asyncio.Queue()
        self.running = False

    async def add_task(self, task):
        await self.queue.put(task)

    async def process_queue(self):
        while True:
            task = await self.queue.get()
            result = await self.execute_task(task)
            await self.notify_completion(task, result)

Integration Example

class RoboticSystem:
    def __init__(self):
        self.parser = TaskParser()
        self.validator = ConstraintValidator(ROBOT_CAPABILITIES)
        self.queue = TaskQueue()
        self.robot = WalkingRobot()  # Our existing walking skeleton

    async def handle_command(self, text_command):
        # Parse the command
        task = self.parser.parse_command(text_command)

        # Validate constraints
        validation = self.validator.validate_task(task)

        if validation['possible']:
            # Add to execution queue
            await self.queue.add_task(task)
            return {
                'status': 'accepted',
                'message': f"Task '{text_command}' queued for execution"
            }
        else:
            return {
                'status': 'rejected',
                'message': validation['reason']
            }

const startTaskMonitoring = (taskId) => {
    // Set up WebSocket connection for real-time updates
    const ws = new WebSocket(`ws://robot-api/task/${taskId}`);

    ws.onmessage = (event) => {
        const update = JSON.parse(event.data);
        updateTaskStatus(update);
    };
};

Implementation Steps

  1. Setup Backend API:
from fastapi import FastAPI, WebSocket

app = FastAPI()
robot_system = RoboticSystem()

@app.post("/command")
async def handle_command(command: str):
    return await robot_system.handle_command(command)

@app.websocket("/ws/tasks/{task_id}")
async def task_status(websocket: WebSocket, task_id: str):
    await websocket.accept()
    while True:
        status = await robot_system.get_task_status(task_id)
        await websocket.send_json(status)
        await asyncio.sleep(0.1)
